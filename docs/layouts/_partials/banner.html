{{- /* Override of the theme's banner that renders only when the site is being viewed at the default (bleeding edge) version root, i.e. on a page that is not under any non-default version's URL path. Configured via site params: params.banner.message - message shown when viewing the default version params.banner.versionWarning - message shown when the default version is the development docs (main) */ -}} {{- $versions := partial "version-list.html" . -}} {{- if and site.Params.banner $versions }} {{- $page := . -}} {{- $isDefault := true -}} {{- with $versions -}} {{- $rel := $page.RelPermalink -}} {{- /* Skip the default entry (index 0): its URL path may equal the site's own path prefix and would match every page. Only a page living under a non-default version's path takes us off the default root. */ -}} {{- range $i, $v := . -}} {{- if gt $i 0 -}} {{- $vPath := (urls.Parse $v.url).Path -}} {{- if and (ne $vPath "/") (strings.HasPrefix $rel $vPath) -}}{{- $isDefault = false -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if $isDefault }}
{{- with partial "custom/banner.html" . -}} {{- . -}} {{- else -}}
{{- $msg := site.Params.banner.message | default "🎉 Welcome! This is a banner message." -}} {{- if and $versions site.Params.banner.versionWarning -}} {{- $default := index $versions 0 -}} {{- if eq $default.name "main" -}} {{- $msg = site.Params.banner.versionWarning -}} {{- end -}} {{- end -}} {{- $msg | .RenderString -}}
{{- end -}}
{{- end -}} {{- end -}}
